// ============================================================================
//  BreezeLink - ESP32 - OUTDOOR node - FALLBACK build (direct WiFi + MQTT)
// ----------------------------------------------------------------------------
//  Reads the DHT and pushes telemetry {t,h,ts,rssi,fw} to the topic
//  bl/{ORG_ID}/{DEVICE_UUID}/telemetry over plaintext MQTT on 1883 (self-hosted
//  EMQX). This node is "outdoor" -> the app/web show it as "Ngoài trời".
//
//  Identical to the indoor node in its backend contract
//  (topic/payload/client-id/LWT).
//
//  IN PRODUCTION use the ESP-NOW build (main-espnow-slave.cpp): the outdoor node
//  sends to the indoor node rather than going online itself. This build is kept as
//  a FALLBACK for when ESP-NOW misbehaves -- it does not depend on the indoor node
//  so it isolates faults very quickly: if this build produces readings and the
//  ESP-NOW one does not, the fault is in the ESP-NOW path rather than in the
//  sensor or the backend.
//      pio run -e esp32-wifi -t upload
// ============================================================================
#include <Arduino.h>
#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>
#include <DHT.h>
#include "config.h"

// The broker runs plaintext on 1883 -> a plain WiFiClient (no TLS), light on RAM.
static WiFiClient   net;
static PubSubClient mqtt(net);
static DHT          dht(DHT_PIN, DHT_TYPE);

static String tTelemetry, tStatus, tCmd;
static void buildTopics() {
  String base = String("bl/") + ORG_ID + "/" + DEVICE_UUID + "/";
  tTelemetry = base + "telemetry";
  tStatus    = base + "status";
  tCmd       = base + "cmd";
}

static void onMessage(char *topic, byte *payload, unsigned int len) {
  Serial.printf("[cmd] %.*s\n", (int)len, (const char *)payload);
}

static void connectWifi() {
  if (WiFi.status() == WL_CONNECTED) return;
  Serial.printf("WiFi -> \"%s\" ", WIFI_SSID);
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.printf(" OK  IP=%s  RSSI=%d dBm\n", WiFi.localIP().toString().c_str(), (int)WiFi.RSSI());
}

static void connectMqtt() {
  String cid = String("breezelink_") + DEVICE_UUID;  // = mqtt_naming.client_id()
  while (!mqtt.connected()) {
    Serial.print("MQTT ... ");
    if (mqtt.connect(cid.c_str(), MQTT_USERNAME, MQTT_PASSWORD,
                     tStatus.c_str(), 1, true, "offline")) {
      Serial.println("connected");
      mqtt.publish(tStatus.c_str(), "online", true);
      mqtt.subscribe(tCmd.c_str(), 1);
    } else {
      Serial.printf("failed rc=%d (retrying in 2s)\n", mqtt.state());
      delay(2000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  delay(300);
  Serial.println("\n== BreezeLink - ESP32 - OUTDOOR (direct WiFi + MQTT) ==");
  dht.begin();
  buildTopics();
  connectWifi();
  mqtt.setServer(MQTT_HOST, MQTT_PORT);
  mqtt.setCallback(onMessage);
  connectMqtt();
}

static unsigned long lastPub = 0;
void loop() {
  connectWifi();
  if (!mqtt.connected()) connectMqtt();
  mqtt.loop();

  unsigned long now = millis();
  if (lastPub != 0 && now - lastPub < TELEMETRY_MS) return;

  float t = dht.readTemperature();
  float h = dht.readHumidity();
  if (isnan(t) || isnan(h)) {
    // The DHT22's minimum sampling interval is 2s -- wait 3s to let the sensor
    // recover.
    Serial.println("Sensor read error (NaN) - check the DHT wiring/supply");
    delay(3000);
    return;
  }
  lastPub = now;

  JsonDocument doc;
  doc["ts"]   = (uint32_t)(millis() / 1000);
  doc["t"]    = t;
  doc["h"]    = h;
  doc["rssi"] = (int)WiFi.RSSI();
  doc["fw"]   = FW_VERSION;
  char buf[192];
  size_t n = serializeJson(doc, buf);
  bool ok = mqtt.publish(tTelemetry.c_str(), (const uint8_t *)buf, n, false);
  Serial.printf("[telemetry] t=%.1f h=%.0f -> %s\n", t, h, ok ? "sent" : "SEND ERROR");
}
